# ============================================================
# PRACTICA 1 - COMPARTIR ARCHIVOS CON NFS
# VM Server: 10.0.0.215 | VM Cliente: 10.0.0.174
# ============================================================

# --- EN EL SERVIDOR ---

sudo apt update
sudo apt install nfs-kernel-server -y

sudo systemctl status nfs-kernel-server

sudo mkdir -p /OS3

for i in $(seq 1 100); do sudo touch /OS3/Adrian${i}.txt; done

ls /OS3

sudo chown -R nobody:nogroup /OS3
sudo chmod 777 /OS3

echo "/OS3    10.0.0.0/24(rw,sync,no_subtree_check,no_root_squash)" | sudo tee -a /etc/exports

cat /etc/exports

sudo exportfs -a
sudo systemctl restart nfs-kernel-server
sudo exportfs -v

sudo ufw allow from 10.0.0.0/24 to any port nfs
sudo ufw allow 2049
sudo ufw reload

sudo ss -tlnp | grep 2049

# --- EN EL CLIENTE ---

sudo apt update
sudo apt install nfs-common -y

sudo mkdir -p /mnt/OS3
sudo mount 10.0.0.215:/OS3 /mnt/OS3

ls /mnt/OS3

echo "10.0.0.215:/OS3    /mnt/OS3    nfs    defaults    0    0" | sudo tee -a /etc/fstab

cat /etc/fstab

sudo reboot

# Despues del reinicio:
ls /mnt/OS3

sudo apt install samba krb5-config krb5-user winbind -y
# Cuando pregunte el reino Kerberos escribir: SO3.INET

sudo useradd -m lanegracubana
sudo passwd lanegracubana
# Contrasena: 20252185

sudo systemctl stop samba-ad-dc smbd nmbd winbind
sudo systemctl disable samba-ad-dc smbd nmbd winbind

sudo mv /etc/samba/smb.conf /etc/samba/smb.conf.bak

sudo samba-tool domain provision --use-rfc2307 --interactive
# Realm: SO3.INET
# Domain: SO3
# Server Role: dc
# DNS backend: SAMBA_INTERNAL
# DNS forwarder IP: 8.8.8.8
# Administrator password: 20252185

sudo cp /var/lib/samba/private/krb5.conf /etc/krb5.conf

sudo systemctl unmask samba-ad-dc
sudo systemctl enable samba-ad-dc
sudo systemctl start samba-ad-dc

sudo systemctl status samba-ad-dc

sudo samba-tool user create lanegracubana 20252185

sudo samba-tool user list

sudo samba-tool domain level show

host -t A SO3.inet 127.0.0.1
