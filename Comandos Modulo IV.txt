PRACTICA 1 - SERVIDOR HTTP APACHE2
================================================================

sudo apt update
sudo apt install apache2 -y
sudo systemctl start apache2
sudo systemctl enable apache2
sudo systemctl status apache2
sudo mkdir -p /var/www/holamundo
sudo nano /var/www/holamundo/index.html
sudo nano /etc/apache2/sites-available/holamundo.conf
sudo a2ensite holamundo.conf
sudo systemctl reload apache2
cd /etc/apache2
sudo nano ports.conf
sudo mkdir -p /var/www/personal
sudo nano /var/www/personal/index.html
sudo nano /etc/apache2/sites-available/personal.conf
sudo a2ensite personal.conf
sudo systemctl restart apache2
sudo netstat -tlnp | grep apache
sudo apache2ctl -S
