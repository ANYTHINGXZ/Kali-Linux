# PRACTICA 1 - CIFRADO GPG
# ============================================================

sudo apt update
sudo apt install -y gnupg2

gpg --version

mkdir ~/practica_cifrado
cd ~/practica_cifrado
echo "Este es el contenido del archivo secreto" > archivo.txt

gpg --symmetric --cipher-algo AES256 archivo.txt

cat archivo.txt.gpg

cat archivo.txt

rm archivo.txt

cat archivo.txt

gpg --output archivo_descifrado.txt --decrypt archivo.txt.gpg

cat archivo_descifrado.txt


# ============================================================
# PRACTICA 2 - IP TABLES / UFW
# ============================================================

sudo apt update
sudo apt install -y apache2 vsftpd openssh-server

sudo systemctl start apache2
sudo systemctl enable apache2
sudo systemctl start vsftpd
sudo systemctl enable vsftpd
sudo systemctl start ssh
sudo systemctl enable ssh

sudo systemctl status apache2
sudo systemctl status vsftpd
sudo systemctl status ssh

sudo iptables -L -n -v

sudo iptables -A INPUT -p tcp --dport 80 -j DROP
sudo iptables -A INPUT -p tcp --dport 21 -j DROP
sudo iptables -A INPUT -p tcp --dport 22 -j DROP

sudo iptables -L -n -v

sudo iptables -D INPUT -p tcp --dport 80 -j DROP
sudo iptables -D INPUT -p tcp --dport 21 -j DROP
sudo iptables -D INPUT -p tcp --dport 22 -j DROP

sudo iptables -L -n -v

sudo apt install -y ufw

sudo ufw enable

sudo ufw deny 80/tcp
sudo ufw deny 21/tcp
sudo ufw deny 22/tcp

sudo ufw status verbose

sudo ufw delete deny 80/tcp
sudo ufw delete deny 21/tcp
sudo ufw delete deny 22/tcp

sudo ufw allow 80/tcp
sudo ufw allow 21/tcp
sudo ufw allow 22/tcp

sudo ufw status verbose


# ============================================================
# PRACTICA 3 - INSTALACION IDS SNORT
# ============================================================

sudo apt update
sudo apt install -y snort

snort --version

sudo nano /etc/snort/rules/local.rules

alert icmp any any -> $HOME_NET any (msg:"ICMP Ping Detectado"; sid:1000001; rev:1;)
alert tcp any any -> $HOME_NET 21 (msg:"Trafico FTP Puerto 21 Detectado"; sid:1000002; rev:1;)
alert tcp any any -> $HOME_NET 22 (msg:"Trafico SSH Puerto 22 Detectado"; sid:1000003; rev:1;)
alert tcp any any -> $HOME_NET 80 (msg:"Trafico HTTP Puerto 80 Detectado"; sid:1000004; rev:1;)

sudo snort -T -c /etc/snort/snort.conf

sudo snort -A console -q -c /etc/snort/snort.conf -i eth0
